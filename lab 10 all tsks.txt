.model small
.stack 100h
.data
arr db 5 dup(?)
sum db 0
msg1 db 'Enter 5 numbers:$'
msg2 db 13,10,'Sum = $'

.code
main proc
    mov ax,@data
    mov ds,ax

    lea dx,msg1
    mov ah,09h
    int 21h

    mov cx,5
    lea si,arr

input:
    mov ah,01h
    int 21h
    sub al,48

    mov [si],al
    add sum,al

    inc si
    loop input

    lea dx,msg2
    mov ah,09h
    int 21h

    mov dl,sum
    add dl,48
    mov ah,02h
    int 21h

    mov ah,4ch
    int 21h
main endp
end main
